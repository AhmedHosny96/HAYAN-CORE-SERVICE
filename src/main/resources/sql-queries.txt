CREATE TABLE Agent (
    Id BIGINT PRIMARY KEY IDENTITY,
    Name VARCHAR(255),
    ContactPerson VARCHAR(255),
    Country VARCHAR(255),
    City VARCHAR(255),
    ContactEmail VARCHAR(255),
    Status INT,
    TypeId INT, -- Ensuring TypeId matches the datatype of AgentType(Id)
    CreatedAt DATETIME NOT NULL DEFAULT GETDATE(),
    UpdatedAt DATETIME,
    FOREIGN KEY (TypeId) REFERENCES AgentType(Id)
);


CREATE TABLE AgentType (
    Id INT PRIMARY KEY IDENTITY,
    Type VARCHAR(255) UNIQUE,
    CreatedAt DATETIME NOT NULL DEFAULT GETDATE(),
    UpdatedAt DATETIME
);


CREATE TABLE Role (
    Id BIGINT IDENTITY PRIMARY KEY,
    Name VARCHAR(255) NOT NULL UNIQUE,
    Description TEXT,
    Status INT,
    CreatedDate DATETIME NOT NULL,
    UpdatedDate DATETIME,
    CONSTRAINT UC_Role_Name UNIQUE (Name)
);



CREATE TABLE Users (
    Id INT IDENTITY(1,1) PRIMARY KEY,
    Username NVARCHAR(255),
    Email NVARCHAR(255),
    PhoneNumber NVARCHAR(255),
    FullName NVARCHAR(255),
    Password NVARCHAR(255),
    AgentId BIGINT,
    FOREIGN KEY (agentId) REFERENCES Agent(Id),
    RoleId BIGINT,
    FOREIGN KEY (roleId) REFERENCES Role(Id),
    FirstLogin BIT,
    Status BIT,
    CreatedBy NVARCHAR(255),
    CreatedDate DATETIME NOT NULL DEFAULT GETDATE()
);

CREATE TABLE CommissionType (
    Id INT PRIMARY KEY IDENTITY,
    Type NVARCHAR(255) UNIQUE
);

-- Create Commission table
CREATE TABLE Commission (
    Id INT PRIMARY KEY IDENTITY,
    CommissionTypeId INT FOREIGN KEY REFERENCES CommissionType(Id),
    Value FLOAT,
    Agent INT FOREIGN KEY REFERENCES [User](Id),
    Status INT
);
